# Admin trong Portfolio — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Trang `/admin` trong app Portfolio để sửa `site_content` (en, vi, profile, settings), upload CV và xuất bản ngay.

**Architecture:** `app/` tách thành 2 root layout bằng route group: `(site)` (portfolio hiện tại) và `(admin)/admin`.
Admin là client component dùng `@supabase/supabase-js` (anon key + RLS) để đăng nhập/đọc/ghi; một server action kiểm tra
token rồi `revalidateTag("content")`. Form tự sinh từ cấu trúc mặc định trong `lib/content.ts`, logic thuần nằm trong
`lib/merge.ts` và `lib/admin.ts` để test bằng `node:test`.

**Tech Stack:** Next.js 16.1.6 (App Router), React 19.2, Tailwind 4, TypeScript, `@supabase/supabase-js` v2, Node 22 (`node:test`).

**Spec:** `docs/superpowers/specs/2026-11-18-admin-design.md`

## Global Constraints

- Dependency mới duy nhất: `@supabase/supabase-js`. Không thêm framework test, UI kit, form lib, zod.
- Không thêm biến môi trường. Dùng `SUPABASE_URL`, `SUPABASE_ANON_KEY` có sẵn. Xoá `REVALIDATE_SECRET`.
- `lib/merge.ts` và `lib/admin.ts` chỉ được `import type` (để `node --experimental-strip-types` chạy được không cần đuôi `.ts`).
- Không dùng `any` (eslint `@typescript-eslint/no-explicit-any` đang bật).
- Màu theme phải khớp `^[#\w\s().,/%-]+$`.
- UI tiếng Việt, dùng token Tailwind có sẵn: `bg-bg`, `bg-surface`, `text-ink`, `text-muted`, `border-line`, `bg-accent`, `text-on-accent`.
- URL công khai không đổi: `/`, `/s/<scene>`, `/robots.txt`, `/sitemap.xml`, `/api/weather`.

import type { Site } from "./content";

// Pure helpers shared by the server read (lib/site.ts) and the admin page. Type-only imports so
// `node --experimental-strip-types` can run the tests without file extensions.
type Json = unknown;
const isObj = (v: Json): v is Record<string, Json> => typeof v === "object" && v !== null && !Array.isArray(v);

// An empty value of the same shape: the template for array items the admin adds.
export const blank = (d: Json): Json =>
  Array.isArray(d) ? [] : isObj(d) ? Object.fromEntries(Object.entries(d).map(([k, v]) => [k, blank(v)])) : typeof d === "number" ? 0 : typeof d === "boolean" ? false : "";

// Takes `remote` where it has the same type as `def`, keeps `def` otherwise, so a bad row can't crash
// a component. Array items are checked against the first default item; empty default arrays and
// free-form records (theme) accept any primitive values.
export function merge(def: Json, remote: Json): Json {
  if (Array.isArray(def)) {
    if (!Array.isArray(remote)) return def;
    if (def.length === 0) return remote.filter((v) => typeof v === "string" || typeof v === "number");
    const tpl = blank(def[0]);
    return remote.map((r) => merge(tpl, r));
  }
  if (isObj(def)) {
    if (!isObj(remote)) return def;
    if (Object.keys(def).length === 0) return Object.fromEntries(Object.entries(remote).filter(([, v]) => typeof v === "string"));
    return Object.fromEntries(Object.entries(def).map(([k, v]) => [k, k in remote ? merge(v, remote[k]) : v]));
  }
  return typeof remote === typeof def ? remote : def;
}

export type Row = { key: string; data: Json };

// Supabase rows laid over the defaults, exactly as the site shows them.
export function fromRows(def: Site, rows: Row[]): Site {
  const row = (k: string) => rows.find((r) => r.key === k)?.data;
  const settings = merge(def.settings, row("settings")) as Site["settings"];
  // Every section exactly once, in the admin's order; unknown ids dropped, missing ones appended.
  const ids = def.settings.sections.map((d) => d.id);
  const order = settings.sections.filter((s, i, a) => ids.includes(s.id) && a.findIndex((x) => x.id === s.id) === i);
  return {
    content: { en: merge(def.content.en, row("en")), vi: merge(def.content.vi, row("vi")) } as Site["content"],
    profile: merge(def.profile, row("profile")) as Site["profile"],
    settings: { ...settings, sections: [...order, ...def.settings.sections.filter((d) => !order.some((s) => s.id === d.id))] },
  };
}
```

import type { Site } from "./content";

// Pure editing helpers for the admin form. Type-only imports (see lib/merge.ts).
export type Path = (string | number)[];
type Bag = Record<string | number, unknown>;

export const getIn = (o: unknown, p: Path): unknown => p.reduce<unknown>((v, k) => (v as Bag | undefined)?.[k], o);

export function setIn<T>(o: T, p: Path, v: unknown): T {
  if (!p.length) return v as T;
  const [k, ...rest] = p;
  const copy = (Array.isArray(o) ? [...o] : { ...(o as object) }) as Bag;
  copy[k] = setIn(copy[k], rest, v);
  return copy as T;
}

// Applies `f` at `p` under every root: the content tab edits arrays of en and vi together.
export const setAll = (site: Site, roots: Path[], p: Path, f: (v: unknown) => unknown): Site =>
  roots.reduce((s, r) => setIn(s, [...r, ...p], f(getIn(s, [...r, ...p]))), site);

export type ArrayOp = { type: "add"; item: unknown } | { type: "remove"; index: number } | { type: "move"; index: number; to: number };

export function applyOp(arr: unknown[], op: ArrayOp): unknown[] {
  const a = [...arr];
  if (op.type === "add") a.push(op.item);
  else if (op.type === "remove") a.splice(op.index, 1);
  else if (op.to >= 0 && op.to < a.length) a.splice(op.to, 0, ...a.splice(op.index, 1));
  return a;
}

// Empty lines in the one-per-line textareas are not items.
export const clean = <T,>(v: T): T =>
  (Array.isArray(v)
    ? v.filter((x) => x !== "").map(clean)
    : v && typeof v === "object"
      ? Object.fromEntries(Object.entries(v).map(([k, x]) => [k, clean(x)]))
      : v) as T;

const COLOR = /^[#\w\s().,/%-]+$/;

// Arrays of objects (projects, jobs, stats...) must line up between en and vi.
function sameLength(a: unknown, b: unknown, p: string, errs: string[]) {
  if (Array.isArray(a) && Array.isArray(b)) {
    if (typeof a[0] === "object" || typeof b[0] === "object") {
      if (a.length !== b.length) errs.push(`${p}: EN có ${a.length} mục, VI có ${b.length} mục`);
      a.forEach((x, i) => sameLength(x, b[i], `${p}[${i + 1}]`, errs));
    }
  } else if (a && b && typeof a === "object" && typeof b === "object") {
    for (const k of Object.keys(a)) sameLength((a as Bag)[k], (b as Bag)[k], p ? `${p}.${k}` : k, errs);
  }
}

// Everything that would make the site quietly fall back to defaults. Empty list = safe to save.
export function validate(s: Site): string[] {
  const errs: string[] = [];
  if (!URL.canParse(s.profile.site)) errs.push("Hồ sơ: site phải là URL hợp lệ (vd https://hquanpham.id.vn)");
  sameLength(s.content.en, s.content.vi, "", errs);
  for (const lang of ["en", "vi"] as const) {
    const urls = s.content[lang].projects.items.map((p) => p.url.trim());
    urls.forEach((u, i) => {
      if (!u) errs.push(`Dự án ${i + 1} (${lang}): thiếu url`);
      else if (urls.indexOf(u) !== i) errs.push(`Dự án ${i + 1} (${lang}): url trùng với dự án ${urls.indexOf(u) + 1}`);
    });
    s.content[lang].about.stats.forEach((st, i) => {
      if (!Number.isFinite(st.value)) errs.push(`Số liệu ${i + 1} (${lang}): value phải là số`);
    });
  }
  s.content.en.projects.items.forEach((p, i) => {
    const vi = s.content.vi.projects.items[i];
    if (vi && vi.url.trim() !== p.url.trim()) errs.push(`Dự án ${i + 1}: url EN và VI phải giống nhau`);
  });
  for (const mode of ["light", "dark"] as const) {
    for (const [k, v] of Object.entries(s.settings.theme[mode])) if (!COLOR.test(v)) errs.push(`Màu ${k} (${mode}): "${v}" không hợp lệ`);
  }
  return errs;
}
```

type Props = { def: unknown; site: Site; roots: Path[]; path: Path; label: string; update: Update };

// Walks the DEFAULT shape, never the data: unknown fields never show, new fields in lib/content.ts

export default function Field({ def, site, roots, path, label, update }: Props) {
  const at = (r: Path) => getIn(site, [...r, ...path]);

  if (Array.isArray(def) && typeof def[0] === "object") {
    const items = (at(roots[0]) as unknown[] | undefined) ?? [];
    const op = (o: ArrayOp) => update((s) => setAll(s, roots, path, (v) => applyOp((v as unknown[] | undefined) ?? [], o)));
    return (
      <fieldset className={box}>
        <legend className="px-1 font-semibold">{label}</legend>
        {items.map((_, i) => (
          <div key={i} className="grid gap-3 rounded-lg bg-bg p-3">
            <div className="flex items-center gap-2">
              <span className="mr-auto text-sm text-muted">#{i + 1}</span>
              <button type="button" className={btn} disabled={i === 0} aria-label={`Đưa ${label} ${i + 1} lên`} onClick={() => op({ type: "move", index: i, to: i - 1 })}>↑</button>
              <button type="button" className={btn} disabled={i === items.length - 1} aria-label={`Đưa ${label} ${i + 1} xuống`} onClick={() => op({ type: "move", index: i, to: i + 1 })}>↓</button>
              <button type="button" className={btn} onClick={() => confirm(`Xoá ${label} #${i + 1}?`) && op({ type: "remove", index: i })}>Xoá</button>
            </div>
            <Field def={def[0]} site={site} roots={roots} path={[...path, i]} label="" update={update} />
          </div>
        ))}
        <button type="button" className={`${btn} justify-self-start`} onClick={() => op({ type: "add", item: blank(def[0]) })}>+ Thêm</button>
      </fieldset>
    );
  }

  if (def && typeof def === "object" && !Array.isArray(def)) {
    const body = Object.entries(def).map(([k, v]) => <Field key={k} def={v} site={site} roots={roots} path={[...path, k]} label={k} update={update} />);
    return label ? (
      <fieldset className={box}>
        <legend className="px-1 font-semibold">{label}</legend>
        {body}
      </fieldset>
    ) : (
      <div className="grid gap-4">{body}</div>
    );
  }

  return (
    <div className="grid gap-1">
      <span className="text-sm text-muted">{label}</span>
      <div className={roots.length > 1 ? "grid gap-2 md:grid-cols-2" : "grid"}>
        {roots.map((r) => (
          <Leaf
            key={r.join(".")}
            def={def}
            value={at(r)}
            label={roots.length > 1 ? `${label} (${r.at(-1)})` : label}
            onChange={(v) => update((s) => setIn(s, [...r, ...path], v))}
          />
        ))}
      </div>
    </div>
  );
}

function Leaf({ def, value, label, onChange }: { def: unknown; value: unknown; label: string; onChange: (v: unknown) => void }) {
  if (typeof def === "number") {
    return <input type="number" aria-label={label} className={input} value={Number.isFinite(value) ? (value as number) : ""} onChange={(e) => onChange(e.target.valueAsNumber)} />;
  }
  if (typeof def === "boolean") {
    return <input type="checkbox" aria-label={label} className="size-5 accent-accent" checked={value === true} onChange={(e) => onChange(e.target.checked)} />;
  }
  if (Array.isArray(def)) {
    const lines = Array.isArray(value) ? (value as string[]) : [];
    return <textarea aria-label={label} title="Mỗi dòng một mục" className={input} rows={Math.max(2, lines.length)} value={lines.join("\n")} onChange={(e) => onChange(e.target.value.split("\n"))} />;
  }
  const text = typeof value === "string" ? value : "";
  return String(def).length > 80 ? (
    <textarea aria-label={label} className={input} rows={4} value={text} onChange={(e) => onChange(e.target.value)} />
  ) : (
    <input aria-label={label} className={input} value={text} onChange={(e) => onChange(e.target.value)} />
  );
}

"use server";
import { revalidateTag } from "next/cache";
import { CONTENT_TAG } from "@/lib/site";

// Called after a save. Public sign-up is off, so a token Supabase accepts is the admin's.
// Writes are guarded separately by RLS; this only drops the cached read.
export async function publish(token: string): Promise<{ ok: true } | { ok: false; error: string }> {
  const url = process.env.SUPABASE_URL;
  const key = process.env.SUPABASE_ANON_KEY;
  if (!url || !key) return { ok: false, error: "thiếu SUPABASE_URL/SUPABASE_ANON_KEY" };
  const res = await fetch(`${url}/auth/v1/user`, { headers: { apikey: key, Authorization: `Bearer ${token}` }, cache: "no-store" });
  if (!res.ok) return { ok: false, error: "phiên đăng nhập không hợp lệ" };
  revalidateTag(CONTENT_TAG, { expire: 0 });
  return { ok: true };
}

"use client";
import { createClient, type Session, type SupabaseClient } from "@supabase/supabase-js";
import { useEffect, useMemo, useState } from "react";
import { content, profile, settings, type Site } from "@/lib/content";
import { fromRows, type Row } from "@/lib/merge";
import { clean, validate } from "@/lib/admin";
import Field, { input, type Update } from "./Field";
import { CvUpload, Sections, Theme } from "./Extras";
import { publish } from "./actions";

const defaults: Site = { content, profile, settings };
const tabs = { content: "Nội dung", profile: "Hồ sơ", settings: "Cài đặt" } as const;
type Tab = keyof typeof tabs;
type Msg = { kind: "ok" | "err"; text: string[] };
const btn = "rounded-lg border border-line px-3 py-2 text-sm disabled:opacity-40";

export default function Admin({ url, anonKey }: { url: string; anonKey: string }) {
  const supabase = useMemo(() => createClient(url, anonKey), [url, anonKey]);
  const [session, setSession] = useState<Session | null>(null);
  const [ready, setReady] = useState(false);
  const [site, setSite] = useState<Site | null>(null);
  const [dirty, setDirty] = useState(false);
  const [tab, setTab] = useState<Tab>("content");
  const [msg, setMsg] = useState<Msg | null>(null);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      setSession(data.session);
      setReady(true);
    });
    const { data } = supabase.auth.onAuthStateChange((_event, s) => setSession(s));
    return () => data.subscription.unsubscribe();
  }, [supabase]);

  // Load once. If the session drops and comes back, `site` (and the unsaved edits) stays.
  useEffect(() => {
    if (!session || site) return;
    supabase
      .from("site_content")
      .select("key,data")
      .then(({ data, error }) => {
        if (error) setMsg({ kind: "err", text: [`Không tải được dữ liệu: ${error.message}. Tải lại trang để thử lại.`] });
        else setSite(fromRows(defaults, (data ?? []) as Row[]));
      });
  }, [session, site, supabase]);

  useEffect(() => {
    if (!dirty) return;
    const warn = (e: BeforeUnloadEvent) => e.preventDefault();
    addEventListener("beforeunload", warn);
    return () => removeEventListener("beforeunload", warn);
  }, [dirty]);

  const update: Update = (f) => {
    setSite((s) => (s ? f(s) : s));
    setDirty(true);
  };

  async function save() {
    if (!site) return;
    const data = clean(site);
    const errs = validate(data);
    if (errs.length) return setMsg({ kind: "err", text: errs });
    setSaving(true);
    setMsg(null);
    const rows = [
      { key: "en", data: data.content.en },
      { key: "vi", data: data.content.vi },
      { key: "profile", data: data.profile },
      { key: "settings", data: data.settings },
    ];
    const { error } = await supabase.from("site_content").upsert(rows);
    if (error) {
      setSaving(false);
      return setMsg({ kind: "err", text: [`Lưu thất bại: ${error.message}`] });
    }
    setSite(data);
    setDirty(false);
    const token = (await supabase.auth.getSession()).data.session?.access_token ?? "";
    const res = await publish(token);
    setSaving(false);
    setMsg(
      res.ok
        ? { kind: "ok", text: [`Đã lưu và xuất bản lúc ${new Date().toLocaleTimeString("vi")}.`] }
        : { kind: "err", text: [`Đã lưu nhưng chưa xuất bản (${res.error}). Bấm Lưu & xuất bản lần nữa.`] },
    );
  }

  function restore() {
    if (!confirm(`Đặt tab "${tabs[tab]}" về mặc định? Chỉ ghi khi bấm Lưu & xuất bản.`)) return;
    update((s) => (tab === "content" ? { ...s, content: defaults.content } : tab === "profile" ? { ...s, profile: defaults.profile } : { ...s, settings: defaults.settings }));
  }

  async function signOut() {
    if (dirty && !confirm("Có thay đổi chưa lưu. Vẫn đăng xuất?")) return;
    await supabase.auth.signOut();
    setSite(null);
    setDirty(false);
  }

  if (!ready) return <main className="p-6 text-muted">Đang tải…</main>;
  if (!session) return <Login supabase={supabase} />;

  return (
    <main className="mx-auto grid max-w-5xl gap-4 p-4 md:p-6">
      <header className="sticky top-0 z-10 flex flex-wrap items-center gap-2 border-b border-line bg-bg py-3">
        <div role="tablist" className="flex gap-1">
          {(Object.keys(tabs) as Tab[]).map((t) => (
            <button key={t} role="tab" aria-selected={tab === t} className={`${btn} ${tab === t ? "bg-surface font-semibold" : ""}`} onClick={() => setTab(t)}>
              {tabs[t]}
            </button>
          ))}
        </div>
        <span className="ml-auto text-sm text-muted">{dirty ? "Chưa lưu" : ""}</span>
        <button className={btn} onClick={restore} disabled={!site}>Khôi phục mặc định</button>
        <button className={`${btn} border-accent bg-accent font-semibold text-on-accent`} onClick={save} disabled={!site || saving}>
          {saving ? "Đang lưu…" : "Lưu & xuất bản"}
        </button>
        <button className={btn} onClick={signOut}>Đăng xuất</button>
      </header>

      {msg && (
        <div role={msg.kind === "err" ? "alert" : "status"} className={`rounded-xl border p-3 text-sm ${msg.kind === "err" ? "border-red-400 text-red-300" : "border-accent text-accent"}`}>
          <ul className="grid gap-1">{msg.text.map((t) => <li key={t}>{t}</li>)}</ul>
        </div>
      )}

      {!site ? (
        <p className="text-muted">Đang tải dữ liệu…</p>
      ) : tab === "content" ? (
        <Field def={defaults.content.en} site={site} roots={[["content", "en"], ["content", "vi"]]} path={[]} label="" update={update} />
      ) : tab === "profile" ? (
        <>
          <Field def={defaults.profile} site={site} roots={[["profile"]]} path={[]} label="" update={update} />
          <CvUpload supabase={supabase} update={update} />
        </>
      ) : (
        <>
          <Sections site={site} update={update} />
          <Theme site={site} update={update} />
          <Field def={defaults.settings.seo} site={site} roots={[["settings", "seo"]]} path={[]} label="seo" update={update} />
        </>
      )}
    </main>
  );
}

function Login({ supabase }: { supabase: SupabaseClient }) {
  const [err, setErr] = useState("");
  const [busy, setBusy] = useState(false);
  async function submit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const f = new FormData(e.currentTarget);
    setBusy(true);
    setErr("");
    const { error } = await supabase.auth.signInWithPassword({ email: String(f.get("email")), password: String(f.get("password")) });
    setBusy(false);
    if (error) setErr(error.message);
  }
  return (
    <main className="grid min-h-dvh place-items-center p-4">
      <form onSubmit={submit} className="grid w-full max-w-sm gap-3 rounded-2xl border border-line bg-surface p-6">
        <h1 className="text-xl font-semibold">Admin</h1>
        <label className="grid gap-1 text-sm">Email<input name="email" type="email" required autoComplete="username" className={input} /></label>
        <label className="grid gap-1 text-sm">Mật khẩu<input name="password" type="password" required autoComplete="current-password" className={input} /></label>
        {err && <p role="alert" className="text-sm text-red-300">{err}</p>}
        <button disabled={busy} className="rounded-lg bg-accent px-3 py-2 font-semibold text-on-accent disabled:opacity-40">{busy ? "Đang đăng nhập…" : "Đăng nhập"}</button>
      </form>
    </main>
  );
}
